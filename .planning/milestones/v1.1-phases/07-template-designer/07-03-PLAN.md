```typescript
export interface FormField {
  id: string;
  type: 'text' | 'textarea' | 'radio' | 'checkbox' | 'date' | 'phone' | 'signature';
  label: string;
  required: boolean;
  placeholder?: string;
  options?: string[];
  sort: number;
}

export interface Template {
  id: number; name: string; description: string | null;
  schema: FormField[]; schemaVersion: number;
  status: 'DRAFT' | 'PUBLISHED' | 'OFFLINE';
  creatorId: number; creator: { id: number; realName: string };
  createdAt: string; updatedAt: string;
}

export const useTemplateStore = defineStore('template', {
  state: () => ({
    current: null as Template | null,
    selectedFieldId: null as string | null,
  }),
  getters: {
    selectedField(s): FormField | null { ... }
  },
  actions: {
    fetchOne(id: number): Promise<Template>,
    update(id: number, payload): Promise<Template>,
    changeStatus(id: number, action: 'publish' | 'offline'): Promise<Template>,
    selectField(fieldId: string | null): void,
  },
});
```

  ```typescript
  const GROUP_NAME = 'designer';
  // For each group, create a ref and useDraggable:
  useDraggable(basicRef, basicFields, {
    group: { name: GROUP_NAME, pull: 'clone', put: false },
    sort: false,
    clone: (item: FieldTypeDef) => ({
      id: crypto.randomUUID(),
      type: item.type,
      label: item.label,
      required: false,
      placeholder: item.defaultProps.placeholder ?? '',
      options: item.defaultProps.options ? [...item.defaultProps.options] : undefined,
      sort: 0,
    }),
  });
  ```
