  <action>
执行以下 curl 命令验证后端保护（需先获取 admin token）：

```bash
# 获取 token
TOKEN=$(curl -s -X POST http://localhost:3000/api/v1/auth/login \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"admin123"}' | node -e "process.stdin.on('data',d=>process.stdout.write(JSON.parse(d).accessToken))")

# 1. 验证 ADMIN 角色不可删除（假设 ADMIN 角色 id=1）
curl -s -X DELETE -H "Authorization: Bearer $TOKEN" http://localhost:3000/api/v1/roles/1
# 预期: {"code":"BIZ_ERROR","message":"系统角色不可删除"} + HTTP 400

# 2. 验证 ADMIN 权限不能清空
curl -s -X PUT -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d '{"permissionIds":[]}' http://localhost:3000/api/v1/roles/1/permissions
# 预期: {"code":"BIZ_ERROR","message":"ADMIN 角色不能清空所有权限"} + HTTP 400

# 3. 验证角色列表返回 _count.users
curl -s -H "Authorization: Bearer $TOKEN" http://localhost:3000/api/v1/roles | node -e "process.stdin.on('data',d=>{const r=JSON.parse(d);console.log(r.map(x=>x.name+': users='+x._count?.users))})"
# 预期: 每个角色显示 users=N

# 4. p95 性能测试（10 次取第 9 个值）
for i in $(seq 1 10); do
  curl -s -o /dev/null -w "%{time_total}\n" \
    -H "Authorization: Bearer $TOKEN" \
    http://localhost:3000/api/v1/roles
done | sort -n | tail -2 | head -1
# 预期: < 0.500 (500ms)
```
