    import { BizError } from '../../utils/errors';
    import type { ChannelPushStatus } from '@prisma/client';

    const ALLOWED: Record<ChannelPushStatus, ReadonlyArray<ChannelPushStatus>> = {
      PENDING:   ['CANCELLED', 'APPROVED', 'REJECTED'],
      APPROVED:  [],
      REJECTED:  [],
      CANCELLED: [],
    };

    export function assertChannelPushTransition(from: ChannelPushStatus, to: ChannelPushStatus): void {
      const allowed = ALLOWED[from] ?? [];
      if (!allowed.includes(to)) {
        throw new BizError('CHANNEL_PUSH_ILLEGAL_TRANSITION', `不允许从 ${from} 切换到 ${to}`, 422);
      }
    }

    export function isPartnerMutable(status: ChannelPushStatus): boolean {
      return status === 'PENDING';
    }
    ```

    Note: APPROVED/REJECTED transitions are reserved for Phase 35 (recipient review). Partner-side service in this plan only triggers PENDING -> CANCELLED.

    import { prisma } from '../../plugins/prisma';
    import type { DuplicateHint } from './channel-push.constants';

    export async function findChannelPushDuplicates(input: {
      channelPartnerId: number;
      studentName: string;
      studentPhone: string;
      excludeId?: number;
    }): Promise<DuplicateHint[]> {
      const rows = await prisma.channelPush.findMany({
        where: {
          channelPartnerId: input.channelPartnerId,
          studentName:      input.studentName,
          studentPhone:     input.studentPhone,
          ...(input.excludeId ? { id: { not: input.excludeId } } : {}),
        },
        select: { id: true, studentName: true, studentPhone: true, status: true, submittedAt: true },
        orderBy: { submittedAt: 'desc' },
        take: 10,
      });
      return rows;
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser -> /api/v1/channel-push | CHANNEL_PARTNER clients submit/edit/cancel; ownership re-checked on every mutation; status guards enforce PENDING-only mutability. |
| route body -> Prisma | Strict TypeBox; `channelPartnerId` and `recipientUserId` derived server-side from JWT + ChannelPartnerProfile snapshot, never from body. |
| dedup query -> Prisma | Partner-scoped (`channelPartnerId: currentUser.id`) so cross-partner enumeration is impossible. |
| file upload -> filesystem | MIME/size/count enforced; `getSafeChannelPushStoredName` rewrites filename; `resolveSafeChannelPushPath` rejects traversal; bound to `CHANNEL_PUSH_UPLOAD_DIR` env. |
| cancel/edit -> downstream modules | NO calls to VisitRecord, ApprovalApplication, Notification creation paths. REVIEW-06 invariant. |
