  <action>
    Update PropertyEditor.vue to compute the maximum colSpan for the selected field based on its row context.

    Add import:
    ```typescript
    import { remainingCols } from './composables/gridUtils';
    ```

    Add computed for maxColSpan:
    ```typescript
    const maxColSpan = computed(() => {
      if (!field.value || !store.current?.schema) return 12;
      const schema = store.current.schema as SchemaV2;
      for (const item of schema.items) {
        if (item.type === 'row') {
          const idx = item.fields.findIndex(f => f.id === field.value!.id);
          if (idx !== -1) {
            const othersSpan = item.fields
              .filter(f => f.id !== field.value!.id)
              .reduce((s, f) => s + f.colSpan, 0);
            return 12 - othersSpan;
          }
        } else if (item.type === 'group') {
          for (const row of item.rows) {
            const idx = row.fields.findIndex(f => f.id === field.value!.id);
            if (idx !== -1) {
              const othersSpan = row.fields
                .filter(f => f.id !== field.value!.id)
                .reduce((s, f) => s + f.colSpan, 0);
              return 12 - othersSpan;
            }
          }
        }
      }
      return 12;
    });
    ```

    Add import for SchemaV2 type:
    ```typescript
    import type { SchemaV2 } from 'src/types/schema';
    ```

    Update the QSlider template — change static `:max="12"` to dynamic `:max="maxColSpan"`:
    ```html
    <q-slider
      v-model="field.colSpan"
      :min="1"
      :max="maxColSpan"
      :step="1"
      label
      :label-value="field.colSpan + ' / 12'"
      color="primary"
    />
    ```

    Per D-09: dual-channel sync — slider and canvas resize handle both write to the same reactive `field.colSpan`. No watchers needed. The slider's `:max` now reflects row context, preventing overflow (per Pitfall 7 from RESEARCH.md).
