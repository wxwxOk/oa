export type FieldType = 'text' | 'textarea' | 'radio' | 'checkbox' | 'date' | 'phone' | 'signature';

export interface SchemaField {
  id: string;
  type: FieldType;
  label: string;
  required: boolean;
  colSpan: number;
  placeholder?: string;
  options?: string[];
}

export interface SchemaRow { type: 'row'; fields: SchemaField[]; }
export interface SchemaGroup { type: 'group'; title: string; rows: SchemaRow[]; }
export interface SchemaDynamicTable { type: 'dynamic-table'; label: string; colSpan: number; columns: Array<{ key: string; label: string; type: FieldType; width?: number }>; }
export type SchemaItem = SchemaRow | SchemaGroup | SchemaDynamicTable;
export interface SchemaV2 { version: 2; items: SchemaItem[]; }

export function flattenFields(schema: SchemaV2): SchemaField[];
export function createEmptySchema(): SchemaV2;
```

创建 `frontend/src/components/renderer/FieldRenderer.vue`（per D-07）：

Props: `field: SchemaField`, `mode: 'designer' | 'fill' | 'print'`, `modelValue?: any`
Emits: `update:modelValue`
Expose: `validate`, `saveSignature`, `sigRef`

模板结构（三模式分支）：

1. 外层 div `.field-renderer` + `:class="'mode-' + mode"`
2. Label 区域：`field.label` + 红色星号（`v-if="field.required && mode !== 'print'"`）

3. Print 模式（`v-if="mode === 'print'"`）：
   - text/textarea/phone/date：plain text `modelValue`，空值显示 "—"
   - radio：直接显示 modelValue 文本
   - checkbox：`Array.isArray(modelValue) ? modelValue.join('、') : '—'`
   - signature：`<img :src="modelValue" style="max-height: 80px; object-fit: contain" />` 或 "—"

4. Designer 模式（`v-else-if="mode === 'designer'"`）：
   - text/phone：QInput outlined dense disabled `:placeholder="field.placeholder"`
   - textarea：QInput outlined dense disabled type="textarea"
   - radio：QOptionGroup type="radio" disabled `:options="mapOptions(field.options)"` `:model-value="null"`
   - checkbox：QOptionGroup type="checkbox" disabled `:options="mapOptions(field.options)"` `:model-value="[]"`
   - date：QInput outlined dense disabled placeholder="请选择日期"，append slot 含 calendar_today icon
   - signature：div "签名区域"，color var(--oa-text-tertiary)

5. Fill 模式（`v-else`）：
   - 复用 FormFieldRenderer.vue 的完整逻辑（QInput rules, QOptionGroup v-model, QDate popup, phone mask, SignatureField）
   - requiredRule: `(v: string) => !!v?.trim() || '此项为必填'`
   - mapOptions: `(opts?: string[]) => (opts ?? []).map(o => ({ label: o, value: o }))`
   - radio/checkbox/signature 的手动校验逻辑（radioError, checkboxError, sigError ref）
   - expose validate + saveSignature 方法

CSS（scoped）：

.print-value.empty { color: var(--oa-text-tertiary); }
.signature-placeholder {
  height: 80px; border: 1px dashed var(--oa-border); border-radius: 4px;
  display: flex; align-items: center; justify-content: center;
  color: var(--oa-text-tertiary); font-size: 14px;
}
```
