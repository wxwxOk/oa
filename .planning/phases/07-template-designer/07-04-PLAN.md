This component has two modes:
- **preview mode** (in designer canvas): static placeholder, not interactive
- **interactive mode** (for future fill page): actual signature_pad canvas

Props:
```typescript
const props = withDefaults(defineProps<{
  preview?: boolean;
  modelValue?: string; // base64 PNG data URL
}>(), { preview: true, modelValue: '' });

const emit = defineEmits<{
  'update:modelValue': [value: string];
}>();
```

Implementation:
```typescript
import { ref, onMounted, onBeforeUnmount, watch } from 'vue';
import SignaturePad from 'signature_pad';

const canvasRef = ref<HTMLCanvasElement | null>(null);
let pad: SignaturePad | null = null;

onMounted(() => {
  if (props.preview || !canvasRef.value) return;
  canvasRef.value.width = 400;
  canvasRef.value.height = 200;
  pad = new SignaturePad(canvasRef.value, {
    penColor: '#000',
    backgroundColor: '#fff',
    minWidth: 0.5,
    maxWidth: 2.5,
  });
  // Load existing signature if modelValue provided
  if (props.modelValue) {
    pad.fromDataURL(props.modelValue);
  }
});

function clear() {
  pad?.clear();
  emit('update:modelValue', '');
}

function save() {
  if (!pad || pad.isEmpty()) return '';
  const data = pad.toDataURL('image/png');
  emit('update:modelValue', data);
  return data;
}

onBeforeUnmount(() => { pad?.off(); });

defineExpose({ clear, save, isEmpty: () => pad?.isEmpty() ?? true });
```

- Interactive mode: `<canvas ref="canvasRef">` with same dimensions + "清除签名" button below (`q-btn flat dense size="sm" label="清除签名" @click="clear"`)
