export const authGuard = (requiredPerm?: string) =>
  new Elysia({ name: `auth-guard-${requiredPerm ?? 'any'}` })
    .derive({ as: 'scoped' }, async ({ accessJwt, headers }: any) => {
      // JWT 验证 + 权限检查
      if (requiredPerm && !roleCodes.includes('ADMIN') && !permCodes.has(requiredPerm)) {
        throw forbidden(`缺少权限: ${requiredPerm}`);
      }
      return { currentUser: { id, username, realName, roleCodes, permissions } };
    });

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → API (submission list/detail) | 认证用户请求提交数据，需验证权限 |
| client → API (form-stats) | 认证用户请求统计数据，需验证权限 |
| query params → Prisma | 用户输入的筛选参数传入数据库查询 |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-09-01 | Elevation of Privilege | submission.route.ts | mitigate | authGuard('form:submission:list') 模块级鉴权，无权限用户无法访问任何端点 |
| T-09-02 | Elevation of Privilege | form-stats.route.ts | mitigate | authGuard('form:stats:view') 独立权限码，统计数据与提交数据权限分离 |
| T-09-03 | Tampering | query params | mitigate | Prisma 参数化查询，不拼接 SQL；Number() 转换 templateId/sharerId 防止注入 |
| T-09-04 | Information Disclosure | submission detail | mitigate | 详情 API 通过 select 限制返回字段，不暴露 User.password 等敏感字段 |
| T-09-05 | Denial of Service | form-stats groupBy | accept | 统计 API 接受 dateFrom/dateTo 限制查询范围；当前数据量小，可接受全表扫描风险 |
</threat_model>
