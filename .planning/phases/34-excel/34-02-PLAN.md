Required script (Composition API, `<script setup lang="ts">`):

```ts
import { computed, ref } from 'vue';
import { useQuasar } from 'quasar';
import * as XLSX from 'xlsx';
import { parseChannelPushImportRows } from 'src/components/channel-push/channelPushImport';
import { useChannelPushStore } from 'src/stores/channelPush';
import { useResponsive } from 'src/composables/useResponsive';
import type {
  ChannelPushBatchImportFailedRow,
  ChannelPushDuplicateHint,
  ChannelPushImportPreview,
} from 'src/types/channelPush';

const props = defineProps<{ modelValue: boolean }>();
const emit = defineEmits<{
  (e: 'update:modelValue', value: boolean): void;
  (e: 'duplicates', hints: ChannelPushDuplicateHint[]): void;
}>();

const $q = useQuasar();
const store = useChannelPushStore();
const { isMobile } = useResponsive();

const selectedFile = ref<File | null>(null);
const reading = ref(false);
const preview = ref<ChannelPushImportPreview | null>(null);
const failedRows = ref<ChannelPushBatchImportFailedRow[]>([]);

const importLoading = computed(() => store.importLoading);

async function handleFileChange(file: File | null) {
  failedRows.value = [];
  if (!file) { preview.value = null; return; }
  reading.value = true;
  try {
    const buf = await file.arrayBuffer();
    const wb = XLSX.read(buf, { type: 'array', cellDates: false });
    const sheet = wb.Sheets[wb.SheetNames[0]!];
    const rows = XLSX.utils.sheet_to_json<unknown[]>(sheet, { header: 1, blankrows: false });
    preview.value = parseChannelPushImportRows(rows, file.name);
  } finally {
    reading.value = false;
  }
}

async function handleConfirm() {
  if (!preview.value || preview.value.overLimit) return;
  const rows = preview.value.validRows.map((r) => r.payload);
  const result = await store.batchImport(rows);
  $q.notify({ type: 'positive', message: `已导入 ${result.createdCount}/${result.total} 条推送` });
  failedRows.value = result.failedRows;
  if (result.duplicateHints.length > 0) {
    emit('duplicates', result.duplicateHints);
  }
  if (result.failedRows.length === 0) {
    closeDialog();
  }
}

function closeDialog() {
  selectedFile.value = null;
  preview.value = null;
  failedRows.value = [];
  emit('update:modelValue', false);
}
```

Failed-rows tab template binding must show: `第 {{ preview.validRows[failedRow.index]?.rowNumber ?? '?' }} 行 — {{ failedRow.reason }} ({{ failedRow.code ?? 'UNKNOWN' }})` so the displayed row numbers are the original Excel rows, not 0-based array indices (RESEARCH Pitfall #3).

Constraints:
- MUST use `cellDates: false` in `XLSX.read` (RESEARCH Pitfall #9).
- MUST use `blankrows: false` in `sheet_to_json`.
- MUST emit `duplicates` event with the cross-record `duplicateHints[]` so the parent page can open `ChannelPushDuplicateDialog`.
- MUST NOT call `Notify` for axios errors — only for the local "已导入 N/M" success.
- MUST NOT inline parser logic — all parsing goes through `parseChannelPushImportRows`.
- File `accept` MUST include both `.xlsx` and `.xls` (matching VisitImportDialog).
</action>
