# Phase 34 Plan 01: Batch Import Contracts, Parser, and Backend Endpoint

<objective>
Build the deterministic foundation for Excel 批量推送导入: extend frontend types, create the pure parser/validator for the 8-column header format, implement `batchCreateChannelPushes` service with partial-success + duplicate hint aggregation, and wire `POST /api/v1/channel-push/batch-import` JSON endpoint behind `channelPush:create`. All before any UI lands.
</objective>

<context>
Phase 34 mirrors Phase 22 (v1.3 visit import) structurally but adapts for channel-push: 8 columns instead of 15, no date fields, new backend endpoint (visit phase reused an existing `/visits/import`; channel-push needs `/batch-import` from scratch). Backend `normalizePhone` already enforces strict China-mobile regex — the partial-success path absorbs invalid phones into `failedRows[]` rather than blocking the batch (CONTEXT.md D-07 + RESEARCH.md Pitfall #1).

This plan delivers Wave 0 (Red tests + types) + Wave 1 (parser + service + route Green) so Plan 02 can drop the Pinia action and dialog onto a stable backend.
</context>

import {
  CHANNEL_PUSH_IMPORT_HEADERS,
  MAX_CHANNEL_PUSH_IMPORT_ROWS,
  type ChannelPushImportInvalidRow,
  type ChannelPushImportPreview,
  type ChannelPushImportRowError,
  type ChannelPushImportValidRow,
  type ChannelPushImportDuplicateWarning,
  type ChannelPushWritePayload,
} from 'src/types/channelPush';

const COLUMN_FIELDS: Array<keyof ChannelPushWritePayload> = [
  'studentName',
  'studentPhone',
  'studentAge',
  'studentEducation',
  'studentGender',
  'intentStatus',
  'intentNote',
  'remark',
];

const MAX_NAME_LEN = 64;
const MAX_PHONE_LEN = 32;
const MAX_SHORT_TEXT = 64;
const MAX_LONG_TEXT = 1000;

export function validateChannelPushImportHeaders(actualHeaders: unknown[]) {
  const expected = [...CHANNEL_PUSH_IMPORT_HEADERS];
  const actual = expected.map((_, i) => normalizeHeader(actualHeaders[i]));
  const headerErrors = expected.flatMap((exp, i) => {
    const recv = actual[i] ?? '';
    return recv === exp ? [] : [`第 ${i + 1} 列期望「${exp}」，实际「${recv || '空'}」`];
  });
  return {
    headerValid: headerErrors.length === 0,
    expectedHeaders: expected,
    actualHeaders: actual,
    headerErrors,
  };
}

export function parseChannelPushImportRows(
  rows: unknown[][],
  fileName?: string,
): ChannelPushImportPreview {
  const headerResult = validateChannelPushImportHeaders(rows[1] ?? []);
  const preview: ChannelPushImportPreview = {
    fileName,
    ...headerResult,
    validRows: [],
    invalidRows: [],
    duplicateWarnings: [],
    overLimit: false,
  };

  if (!preview.headerValid) return preview;

  for (let i = 2; i < rows.length; i += 1) {
    const cells = (rows[i] ?? []).map(toCell);
    if (cells.every(isBlank)) continue;
    const rowNumber = i + 1;
    const errors: ChannelPushImportRowError[] = [];
    const payload: ChannelPushWritePayload = { studentName: '', studentPhone: '' };

    COLUMN_FIELDS.forEach((field, idx) => {
      const label = CHANNEL_PUSH_IMPORT_HEADERS[idx];
      const value = cells[idx];
      // ... per-field normalization with length checks and integer guard for studentAge
      // (studentName/studentPhone required; age 1..120 integer or empty; everything else trim/optional)
    });

    if (errors.length === 0) {
      preview.validRows.push({ rowNumber, payload });
    } else {
      preview.invalidRows.push({ rowNumber, rawCells: cells.map(String), errors });
    }
  }

  if (preview.validRows.length > MAX_CHANNEL_PUSH_IMPORT_ROWS) {
    preview.overLimit = true;
  }

  // In-file duplicate warnings keyed by `${studentName}|${studentPhone}`
  const groups = new Map<string, ChannelPushImportValidRow[]>();
  for (const row of preview.validRows) {
    const key = `${row.payload.studentName}|${row.payload.studentPhone}`;
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key)!.push(row);
  }
  for (const [key, rs] of groups) {
    if (rs.length > 1) {
      preview.duplicateWarnings.push({
        key,
        rowNumbers: rs.map((r) => r.rowNumber),
        studentName: rs[0]!.payload.studentName,
        studentPhone: rs[0]!.payload.studentPhone,
      });
    }
  }

  return preview;
}

// helpers: normalizeHeader, toCell, isBlank, normalizeText, normalizeAge ...
```

export async function batchCreateChannelPushes(
  currentUser: ChannelPushActor,
  rows: ChannelPushWriteInput[],
): Promise<{
  createdCount: number;
  total: number;
  failedRows: Array<{ index: number; reason: string; code?: string }>;
  duplicateHints: DuplicateHint[];
}> {
  // Pre-flight: bound recipient required for all rows; fail fast.
  const profile = await prisma.channelPartnerProfile.findUnique({
    where: { userId: currentUser.id },
  });
  if (!profile) {
    throw new BizError('未绑定接收人，请联系管理员', 422, 'CHANNEL_PARTNER_NOT_BOUND');
  }

  const failedRows: Array<{ index: number; reason: string; code?: string }> = [];
  const hintMap = new Map<number, DuplicateHint>();
  let createdCount = 0;

  for (let i = 0; i < rows.length; i += 1) {
    const row = rows[i]!;
    try {
      const result = await createChannelPush(currentUser, row, []);
      createdCount += 1;
      for (const hint of result.duplicateHints) {
        if (!hintMap.has(hint.id)) hintMap.set(hint.id, hint);
      }
    } catch (err) {
      const biz = err as BizError;
      failedRows.push({
        index: i,
        reason: biz?.message ?? 'unknown error',
        code: biz?.code,
      });
    }
  }

  return {
    createdCount,
    total: rows.length,
    failedRows,
    duplicateHints: Array.from(hintMap.values()),
  };
}
```
